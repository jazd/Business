Seed a minimal CA + CAPolicy (not present in Business Static).
!|Execute|INSERT INTO CA (name) SELECT GetEntityName('FitNesse Test Issuing CA') WHERE NOT EXISTS (SELECT 1 FROM CA WHERE name = GetEntityName('FitNesse Test Issuing CA'))|

!|Query|SELECT id FROM CA WHERE name = GetEntityName('FitNesse Test Issuing CA') ORDER BY id DESC LIMIT 1|
|id?|
|>>ca|

!|Execute|INSERT INTO CAPolicy (ca, type) SELECT _:ca::INT, GetWord('Issuing') WHERE NOT EXISTS (SELECT 1 FROM CAPolicy WHERE ca = _:ca::INT AND stop IS NULL)|

!|Query|SELECT id FROM CAPolicy WHERE ca = _:ca::INT AND stop IS NULL ORDER BY id DESC LIMIT 1|
|id?|
|>>capolicy|
