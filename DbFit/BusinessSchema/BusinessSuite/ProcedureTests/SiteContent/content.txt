SiteContent and ListContent

One sentence on one content id. A null-target class applies to every output.
The html target uses a different class token. ClientCulture reads the same
sentence id in another culture.

!|Query|SELECT SetContent('DbFit body')|
|setcontent?|
|>>cid|

!|Query|SELECT GetSentence('DbFit line')|
|getsentence?|
|>>sid|

!|Query|INSERT INTO Sentence (id, culture, value) SELECT _:sid::INT, code, 'Ligne DbFit' FROM Culture WHERE name = 'fr-FR' RETURNING id|
|id|
|<<sid|

!|Query|SELECT SetFormat('DbFit format')|
|setformat?|
|>>fmt|

!|Query|SELECT SetFormatAttribute(_:fmt::INT, 'class', 'shared', NULL)|
|setformatattribute|
|<<fmt|

!|Query|SELECT SetFormatAttribute(_:fmt::INT, 'class', 'htmltok', 'html')|
|setformatattribute|
|<<fmt|

!|Query|SELECT SetContentElement(_:cid::INT, 1, _:fmt::INT, NULL, _:sid::INT, NULL, NULL)|
|setcontentelement|
|<<cid|

|getsite?|
